import time, json
import numpy as np
from mpmath import mp, mpf, mpc, matrix, exp, cos, sqrt, pi, log, digamma, quad, re, zetazero, eigsy, cholesky, inverse
mp.dps=70
w=mpf(2); t0=mpf(10); Delta=mpf(1)
MS=[4,8,12,16,20,24,28,32,36,40]

# ---- prime side ingredients
Nmax=40000
sieve=np.ones(Nmax+1,bool); sieve[:2]=False
for p in range(2,int(Nmax**0.5)+1):
    if sieve[p]: sieve[p*p::p]=False
terms=[]   # (u, c) with c=Lambda(n)/sqrt(n)*exp(-w^2 u^2/4)
for p in np.nonzero(sieve)[0]:
    q=int(p)
    while q<=Nmax:
        u=log(mpf(q)); c=log(mpf(int(p)))/sqrt(mpf(q))*exp(-w*w*u*u/4); terms.append((u,c)); q*=int(p)
print("prime-power terms:",len(terms),"smallest kept weight ~",mp.nstr(terms[-1][1],3),flush=True)
tailbound=exp(-w*w*log(mpf(Nmax))**2/4); print("Gaussian tail at n=Nmax: e^-...=",mp.nstr(tailbound,3),flush=True)
cacheP={}; cachePsi={}
def key(tau): return mp.nstr(abs(tau),30)
def P(tau):
    k=key(tau)
    if k not in cacheP:
        tau=abs(tau); cacheP[k]=sum(c*cos(tau*u) for u,c in terms)
    return cacheP[k]
def Psi(m):
    k=key(m)
    if k not in cachePsi:
        m=abs(m)
        f=lambda r: exp(-(r-m)**2/(w*w))*re(digamma(mpf(1)/4+mpc(0,1)*r/2))
        cachePsi[k]=quad(f,[m-14*w,m,m+14*w])/(2*pi)
    return cachePsi[k]
def hb(r,t): return exp(-(r-t)**2/(2*w*w))+exp(-(r+t)**2/(2*w*w))
# zeros (first 60)
t1=time.time(); gam=[zetazero(k).imag for k in range(1,61)]; print("zeros computed",round(time.time()-t1),"s; gamma_60=",mp.nstr(gam[-1],8),flush=True)
def build(m):
    A=matrix(m,m); Mn=matrix(m,m); Pr=matrix(m,m); AZ=matrix(m,m)
    hi=[hb(mpc(0,0.5),ts[j]).real for j in range(m)]
    for j in range(m):
        for k in range(j,m):
            tj,tk=ts[j],ts[k]
            E1=exp(-(tj-tk)**2/(4*w*w)); E2=exp(-(tj+tk)**2/(4*w*w))
            sp,sm=(tj+tk)/2,(tj-tk)/2
            g0=(w/sqrt(pi))*(E1+E2)

            Mn[j,k]=Mn[k,j]=pole+arch; Pr[j,k]=Pr[k,j]=pr
            A[j,k]=A[k,j]=pole+arch-pr
            z=2*sum(hb(g,tj)*hb(g,tk) for g in gam)
            AZ[j,k]=AZ[k,j]=z
    return A,Mn,Pr,AZ

    json.dump(res,open('hp_res.json','w'))
print("done",flush=True)
