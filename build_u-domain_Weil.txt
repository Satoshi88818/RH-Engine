def spec(kind,a,t):
    """phi(v)= 1_{|v|<=a} sum_s alpha_s exp(i nu_s v) ; kind 'W' Hann packet, 'D' derivative packet (i*omega')"""

def I(mu,a,u):
    mu=np.asarray(mu,dtype=float)

def g_cross(u,sj,sk,a):
    """g_jk(u)=int phi_j(v+u) conj(phi_k(v)) dv for u>=0 (u<=2a)"""
