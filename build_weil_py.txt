import os, json, numpy as np, mpmath as mp, matplotlib
matplotlib.use("Agg"); import matplotlib.pyplot as plt
os.environ.setdefault("WEIL_DPS", "170")
import weil_experiment as WE, weil_form as WF
A = json.load(open('weil_sweepA.json')); Tr = json.load(open('weil_transition.json')); CA = json.load(open('weil_convA.json')); CB = json.load(open('weil_convB.json'))
f, ax = plt.subplots(2, 2, figsize=(12.5, 8.6))
# (a) positivity margin vs L (N = 24)
rows = {}
for r in A: rows[r['L']] = (float(r['min_full']), float(r['min_arch']))
for r in Tr: rows[r['L']] = (float(r['min_full']), float(r['min_arch']))
Ls = sorted(rows); full = np.array([rows[l][0] for l in Ls]); arch = np.array([rows[l][1] for l in Ls])
ax[0, 0].semilogy(Ls, full, 'o-', color='C0', label='full form (primes included): smallest eigenvalue')
pos = arch > 0; ax[0, 0].semilogy(np.array(Ls)[pos], arch[pos], 's--', color='C1', label='archimedean term only: smallest eigenvalue (> 0)')
ax[0, 0].semilogy(np.array(Ls)[~pos], np.abs(arch[~pos]), 'x', color='C3', ms=8, label='archimedean only: |smallest eigenvalue| (NEGATIVE)')
ax[0, 0].axvline(0.5 * np.log(2), color='gray', ls=':'); ax[0, 0].text(0.5 * np.log(2) + 0.01, 1e-7, 'prime 2 enters\n(λ² = 2)', fontsize=7)
ax[0, 0].set(xlabel='L = ln λ  (support [1/λ, λ])', ylabel='smallest eigenvalue of the Weil form (N = 24)', title='Positivity margin: primes are needed once λ² > 2'); ax[0, 0].legend(fontsize=7, loc='lower left')
# (b) eigenvalue ladder L=1.1, N=40
ev = np.array([float(x) for x in CB[0]['eig_full']]); ax[0, 1].semilogy(np.arange(1, len(ev) + 1), ev, 'o-', ms=4)
ax[0, 1].set(xlabel='eigenvalue index (ascending)', ylabel='eigenvalue', title=f'Spectrum at L = 1.1, N = 40: all positive, from {ev[0]:.0e} to {ev[-1]:.1f}')
# (c) Fourier transform of the minimal eigenvector, L=1.1, N=40
L, N = 1.1, 40; W = WF.Weil(L, N); M = W.matrix(); E, Q = mp.eigsy(mp.matrix(M)); order = sorted(range(N), key=lambda i: E[i]); c = [float(Q[i, order[0]]) for i in range(N)]
r = np.linspace(0.02, 60, 6000); gh = WE.ghat(c, L, r); F = gh.real if abs(gh.real).max() > abs(gh.imag).max() else gh.imag
ax[1, 0].semilogy(r, np.abs(F) / np.abs(F).max() + 1e-30, lw=1, color='C0', label='|ĝ_min(r)| (from the arithmetic side only)')
for k, z in enumerate(WE.ZEROS[:9]): ax[1, 0].axvline(z, color='C3', ls=':', lw=1, label='zeta zeros' if k == 0 else None)
ax[1, 0].set(xlabel='r', ylabel='|ĝ_min(r)| / max, log scale', title='Minimal eigenvector on [1/3, 3]: |ĝ| dips at the zeta zeros', xlim=(0, 55), ylim=(1e-14, 2)); ax[1, 0].legend(fontsize=7)
# (d) convergence at L = 0.7
Ns = [r_['N'] for r_ in CA]; mins = [float(r_['min_full']) for r_ in CA]; ax[1, 1].semilogy(Ns, mins, 'o-', color='C0', label='smallest eigenvalue')
ax2 = ax[1, 1].twinx(); nm = []
for r_ in CA:
    sc = np.array(r_['sign_changes']); d = [float(np.min(np.abs(sc - z))) if len(sc) else 9 for z in WE.ZEROS[:10]]; k = 0
    while k < 10 and d[k] < 0.05: k += 1
    nm.append(k)
ax2.plot(Ns, nm, 's--', color='C3', label='leading zeros reproduced (within 0.05)'); ax2.set_ylabel('number of leading zeros reproduced', color='C3'); ax2.set_ylim(0, 10)
ax[1, 1].set(xlabel='number of basis functions N', ylabel='smallest eigenvalue', title='Convergence at L = 0.7'); ax[1, 1].legend(fontsize=7, loc='upper right')
plt.tight_layout(); plt.savefig('rh_weil_archimedean.png', dpi=125); print('ok', nm)
