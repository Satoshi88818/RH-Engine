def analyse(kind,a,K,truez,rmax=45):
    A,G,sp=ccm.build(kind,a,K); ev,evec=eigh((A+A.T)/2,G)
    scale=np.abs(ev).max(); lev=np.log10(np.maximum(np.abs(ev),1e-300))
    # rank gap: count eigenvalues above 1e-9*scale
    nz=ev>1e-9*scale; nneg=ev<-1e-9*scale; null=np.abs(ev)<=1e-9*scale
    V=np.array([np.real(phihat(rg,s,a)) for s in sp])             # (K+1) x len(r)
    Cn=evec[:,null]; F=np.sum((Cn.T@V)**2,axis=0)
    mins=[i for i in range(1,len(rg)-1) if F[i]<F[i-1] and F[i]<F[i+1]]
    # keep deep minima: F small relative to its median
    thr=1e-6*np.median(F)
    deep=[rg[i] for i in mins if F[i]<thr and rg[i]<rmax]
    tz=truez[truez<rmax]
    err=[min(abs(d-tz)) for d in deep]
    miss=[z for z in tz if min(abs(z-np.array(deep)))>0.05] if len(deep) else list(tz)
    print(f"{kind} a={a} (primes<={int(np.exp(2*a))}) K={K}: eig>tol:{int(nz.sum())} eig<-tol:{int(nneg.sum())} |eig|<=tol:{int(null.sum())}; deep minima of F: {len(deep)}; known zeros <{rmax}: {len(tz)}; max distance of minimum to a known zero: {max(err) if err else float('nan'):.4f}; known zeros with no minimum within 0.05: {np.round(miss,3)}")
    return deep
