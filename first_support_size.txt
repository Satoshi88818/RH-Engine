"""Spectrum of the Weil form on [-L, L]: positivity, minuscule eigenvalues, and where the minimal eigenvector's Fourier transform vanishes."""
import os, sys, json, time
import numpy as np, mpmath as mp
from scipy.special import spherical_jn
import weil_form as WF
mp.mp.dps = int(os.environ.get("WEIL_DPS", 50))
ZEROS = np.loadtxt('odlyzko/zeros6')[:60]
def ghat(coeffs, L, r):
    out = np.zeros(len(r), dtype=complex)
    for k, c in enumerate(coeffs): out += c * np.sqrt(2 * (2 * k + 1) * L) * (-1j)**k * spherical_jn(k, r * L)
    return out
def eig(Mat, N):
    E, Q = mp.eigsy(mp.matrix(Mat)); order = sorted(range(N), key=lambda i: E[i]); return [E[i] for i in order], Q, order
def run(L, N):
    W = WF.Weil(L, N); M = W.matrix(); Ma = W.M_arch
    Ef, Qf, of = eig(M, N); Ea, Qa, oa = eig(Ma, N)
    res = dict(L=L, N=N, eig_full=[mp.nstr(e, 10) for e in Ef], eig_arch=[mp.nstr(e, 10) for e in Ea], min_full=mp.nstr(Ef[0], 8), min_arch=mp.nstr(Ea[0], 8), n_negative_arch=int(sum(1 for e in Ea if e < 0)), n_negative_full=int(sum(1 for e in Ef if e < 0)))
    c = [float(Qf[i, of[0]]) for i in range(N)]
    r = np.linspace(0.05, 70, 14000); gh = ghat(c, float(L), r); F = gh.real if abs(gh.real).max() > abs(gh.imag).max() else gh.imag
    s = np.sign(F); idx = np.nonzero(s[:-1] * s[1:] < 0)[0]; cross = r[idx] - F[idx] * (r[idx + 1] - r[idx]) / (F[idx + 1] - F[idx])
    near = [float(np.min(np.abs(ZEROS - x))) for x in cross]
    vals = np.array([abs(ghat(c, float(L), np.array([z]))[0])**2 for z in ZEROS[:8]])
    res.update(sign_changes=[float(x) for x in cross], nearest_zero_dist=near, mean_sq_first8_zeros=float(vals.mean()), mean_sq_overall=float(np.mean(np.abs(F)**2)))
    return res
if __name__ == "__main__":
    out = []

        print(f"L={L}: {r['secs']:.0f}s  min eig (full) = {r['min_full']} (#neg {r['n_negative_full']})   min eig (arch only) = {r['min_arch']} (#neg {r['n_negative_arch']})", flush=True)

        json.dump(out, open(sys.argv[3], 'w'))
