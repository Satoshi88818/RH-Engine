import numpy as np, math
from fractions import Fraction
def legendre(a,p):
    a%=p
    if a==0: return 0
    return 1 if pow(a,(p-1)//2,p)==1 else -1
def elliptic_N1(a,b,p): return 1+sum(1+legendre(x**3+a*x+b,p) for x in range(p))
def nonres(p):
    for n in range(2,p):
        if legendre(n,p)==-1: return n
def N2_poly(coefs,p):
    n=nonres(p)
    def mul(u,v): return ((u[0]*v[0]+n*u[1]*v[1])%p,(u[0]*v[1]+u[1]*v[0])%p)
    def pw(u,e):
        r=(1,0)
        while e:
            if e&1: r=mul(r,u)
            u=mul(u,u); e>>=1
        return r
    tot=1
    for x in [(a,b) for a in range(p) for b in range(p)]:
        v=(0,0); xp=(1,0)
        for c in coefs:
            v=((v[0]+c*xp[0])%p,(v[1]+c*xp[1])%p); xp=mul(xp,x)
        tot+=1+(0 if v==(0,0) else (1 if pw(v,(p*p-1)//2)==(1,0) else -1))
    return tot
def power_sums(L,Kmax):
    d=len(L)-1; s=[d]+[0]*Kmax
    for k in range(1,Kmax+1):
        acc=sum(L[m]*s[k-m] for m in range(1,min(k-1,d)+1)); s[k]=-(acc+(k*L[k] if k<=d else 0))
    return s
def counts(L,q,Kmax): s=power_sums(L,Kmax); return [None]+[q**k+1-s[k] for k in range(1,Kmax+1)]
def mobius(n):
    r=1; m=n; p=2
    while p*p<=m:
        if m%p==0:
            m//=p
            if m%p==0: return 0
            r=-r
        p+=1
    return -r if m>1 else r
