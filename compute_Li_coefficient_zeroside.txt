"""Li coefficients from the zero side: lambda_n = sum_{gamma>0} 4 sin^2(n theta/2), theta = 2 arctan(1/(2 gamma)),
using Odlyzko's 2,001,052 published zeros (gamma <= 1,132,490.66) plus a smooth-density tail integral."""
import json, sys, numpy as np
EG = 0.5772156649015329
zs = np.loadtxt('odlyzko/zeros6'); T = zs[-1]; th = 2 * np.arctan(1 / (2 * zs))
def tail(n):
    # int_T^inf 4 sin^2(n theta(g)/2) * ln(g/2pi)/(2pi) dg  on a log grid, plus analytic remainder
    G = 1e15; x = np.linspace(np.log(T), np.log(G), 200001); g = np.exp(x); f = 4 * np.sin(n * 2 * np.arctan(1 / (2 * g)) / 2)**2 * np.log(g / (2 * np.pi)) / (2 * np.pi) * g
    return float(np.trapezoid(f, x) + n**2 / (2 * np.pi) * (np.log(G / (2 * np.pi)) + 1) / G)
def lam_zero(n): return float(np.sum(4 * np.sin(n * th / 2)**2))
def asym(n): return 0.5 * n * (np.log(n) + EG - 1 - np.log(2 * np.pi))
if __name__ == "__main__":
