import sys, json, time
import numpy as np, mpmath as mp
from scipy.special import spherical_jn
import weil_form as WF
mp.mp.dps = 50
ZEROS = np.loadtxt('odlyzko/zeros6')[:60]
def ghat(coeffs, L, r):
    """Fourier transform of g = sum c_k psi_k at real r (real-valued up to the factor (-i)^k, parity decouples): returns Re/Im part chosen by parity."""
    out = np.zeros(len(r), dtype=complex)
    for k, c in enumerate(coeffs): out += c * np.sqrt(2 * (2 * k + 1) * L) * (-1j)**k * spherical_jn(k, r * L)
    return out
def run(L, N, primes=True):
    W = WF.Weil(L, N); M = W.matrix(primes=True); Ma = W.M_arch
    def eig(Mat):
        A = mp.matrix(Mat); E, Q = mp.eigsy(A); order = sorted(range(N), key=lambda i: E[i]); return [E[i] for i in order], Q, order
    Ef, Qf, of = eig(M); Ea, Qa, oa = eig(Ma)
    res = dict(L=L, N=N, eig_full=[float(e) for e in Ef], eig_arch=[float(e) for e in Ea], min_full=mp.nstr(Ef[0], 8), min_arch=mp.nstr(Ea[0], 8))
    # minimal eigenvector of the full form: where does its Fourier transform vanish?
    c = [float(Qf[i, of[0]]) for i in range(N)]
    r = np.linspace(0.05, 70, 14000); gh = ghat(c, float(L), r); F = gh.real if abs(gh.real).max() > abs(gh.imag).max() else gh.imag
    s = np.sign(F); idx = np.nonzero(s[:-1] * s[1:] < 0)[0]; cross = r[idx] - F[idx] * (r[idx + 1] - r[idx]) / (F[idx + 1] - F[idx])
    near = [(float(x), float(np.min(np.abs(ZEROS - x)))) for x in cross]
    # weight of the minimal eigenvector's Fourier transform on the first zeros vs between them
    vals = np.array([abs(ghat(c, float(L), np.array([z]))[0])**2 for z in ZEROS[:8]]); typ = np.mean(np.abs(F)**2)
    res.update(sign_changes=[float(x) for x in cross], nearest_zero_dist=[d for _, d in near], mean_sq_at_first8_zeros=float(vals.mean()), mean_sq_overall=float(typ))
    return res
