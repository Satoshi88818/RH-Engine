"""Window-by-window test of the small-gap (extreme-value) law that underlies  t(x) ~ x^(-2/3):
   for windows of N0 consecutive zeros, u = N0*F(nu_min) should be ~Exp(1), F(x) = (pi^2/9)x^3 - (2pi^4/225)x^5,
   so the window's first-collision time  tau = gap_min^2/2 = (2 pi^2/L^2) nu_min^2  scales like N0^(-2/3)/L^2."""
import json, os, sys
import numpy as np
from scipy.stats import kstest
from scipy.optimize import brentq
LD = np.longdouble
F = lambda x: (np.pi**2 / 9) * x**3 - (2 * np.pi**4 / 225) * x**5
N0S = [100, 300, 1000, 3000, 10000, 30000, 100000]
QS = (0.01, 0.05, 0.50)

def norm_gaps(pos):
    pos = np.asarray(pos)
    gaps = np.diff(pos).astype(float); mid = (0.5 * (pos[1:] + pos[:-1])).astype(float)
    L = np.log(mid / (2 * np.pi)); return gaps * L / (2 * np.pi), gaps, L

def nu_quantile(N0, q):
    return brentq(lambda x: N0 * F(x) + np.log(1 - q), 1e-9, 0.6)

def analyse_arrays(name, nu, gaps, L, t0, minw=20):
    out = dict(name=name, n=len(nu) + 1, L=float(L.mean()), t0=float(t0), windows=[])
    for N0 in N0S:
        m = len(nu) // N0
        if m < minw: continue
        Wn = nu[:m * N0].reshape(m, N0); numin = Wn.min(axis=1); u = N0 * F(numin)
        Lw = L[:m * N0].reshape(m, N0).mean(axis=1)
        ks = kstest(u, 'expon')
        cov = {str(q): float((numin >= nu_quantile(N0, q)).mean()) for q in QS[:2]}
        tau = (gaps[:m * N0].reshape(m, N0).min(axis=1))**2 / 2
        out['windows'].append(dict(N0=N0, m=m, ks_p=float(ks.pvalue), ks_stat=float(ks.statistic), mean_u=float(u.mean()),
                                   median_Y=float(np.median(numin) * N0 ** (1 / 3)), coverage=cov,
                                   median_tau=float(np.median(tau)), tau_pred_median=float(2 * np.pi**2 / np.mean(Lw)**2 * nu_quantile(N0, 0.5)**2)))
    return out

def analyse(name, pos):
    nu, gaps, L = norm_gaps(pos)
    return analyse_arrays(name, nu, gaps, L, float(pos[0]))

if __name__ == "__main__":
    res = []
    tab = np.loadtxt('odlyzko/zeros6')
    for lo, hi in [(1e3, 1e4), (1e4, 1e5), (1e5, 1e6), (1e6, 1.2e6)]:
        res.append(analyse(f"Odlyzko zeros6 [{lo:.0e},{hi:.0e})", tab[(tab >= lo) & (tab < hi)]))
    res.append(analyse("Odlyzko zeros3 (~2.7e11)", np.array(np.loadtxt('odlyzko/zeros3', skiprows=7), dtype=LD) + LD(267653395647)))
    for name in ("1e7", "1e8", "1e9", "1e10", "1e11"):
        f = f"pos_{name}.npy"
        if os.path.exists(f): res.append(analyse(f"mine {name}", np.load(f)))
    json.dump(res, open('window_test.json', 'w'), indent=1)
    for r in res:
        print(f"\n{r['name']}  (n={r['n']}, <ln(t/2pi)>={r['L']:.2f})")
        for w in r['windows']:
            print(f"  N0={w['N0']:6d} windows={w['m']:5d} KS p={w['ks_p']:.3f} mean(u)={w['mean_u']:.3f} medianY={w['median_Y']:.3f} cov95={w['coverage']['0.05']:.3f} cov99={w['coverage']['0.01']:.3f}")
