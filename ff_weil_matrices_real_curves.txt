def build(L,q,K):
    g2=len(L)-1; s=power_sums(L,q,K+1)
    P=[mpf(g2)]+[mpf(s[k])/msqrt(mpf(q))**k for k in range(1,K+2)]
    T=matrix(K+1,K+1)
    for j in range(K+1):
        for l in range(K+1): T[j,l]=P[abs(j-l)]
    return T,s
def exactT(L,q,K):
    s=power_sums(L,q,K+1); n=len(L)-1; s[0]=n
    return [[q**min(j,l)*s[abs(j-l)] for l in range(K+1)] for j in range(K+1)]
def nullvec_ok(Tp,L,shift):

    return all(sum(Tp[i][j]*u[j] for j in range(len(u)))==0 for i in range(len(Tp)))
