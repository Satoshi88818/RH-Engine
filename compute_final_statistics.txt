import json, numpy as np
from scipy.optimize import brentq
import window_test as W
LD = np.longdouble
F = W.F
def datasets():
    tab = np.loadtxt('odlyzko/zeros6')
    for lo, hi in [(1e3, 1e4), (1e4, 1e5), (1e5, 1e6), (1e6, 1.2e6)]:
        yield f"zeros6 [{lo:.0e},{hi:.0e}) (published)", tab[(tab >= lo) & (tab < hi)]
    yield "zeros3 ~2.7e11 (published)", np.array(np.loadtxt('odlyzko/zeros3', skiprows=7), dtype=LD) + LD(267653395647)
    for n in ("1e7", "1e8", "1e9", "1e10", "1e11"): yield f"mine {n}", np.load(f"pos_{n}.npy")
rows = []
for name, pos in datasets():
    nu, gaps, L = W.norm_gaps(pos); N = len(nu)
    numin = nu.min(); nu_med = brentq(lambda x: N * F(x) - np.log(2), 1e-9, 0.6)
    i = int(np.argmin(gaps)); tpos = float(0.5 * (pos[i] + pos[i + 1]))
    r = dict(name=name, N=N, L=float(L.mean()), nu_min=float(numin), nu_med=float(nu_med), P=float(1 - np.exp(-N * F(numin))),
             gap_min=float(gaps.min()), at=tpos, tau_obs=float(gaps.min()**2 / 2), tau_pred=float(2 * np.pi**2 / L.mean()**2 * nu_med**2),
             small={str(q): [int((nu < q).sum()), float(N * F(q))] for q in (0.05, 0.1, 0.2)})
    rows.append(r)
    print(f"{name:32s} N={N:8d} L={r['L']:5.2f} nu_min={numin:.4f} (GUE med {nu_med:.4f}, P={r['P']:.2f}) gap_min={gaps.min():.5f} at {tpos:.6g} tau_obs={r['tau_obs']:.2e} tau_pred={r['tau_pred']:.2e} | "
          + " ".join(f"nu<{q}: {v[0]}/{v[1]:.0f}" for q, v in r['small'].items()))
# pooled coverage and exponent
wt = json.load(open('window_test.json'))
tot = {"0.05": [0, 0.0], "0.01": [0, 0.0]}; slopes = []
for d in wt:
    for w in d['windows']:
        for q in tot: tot[q][0] += w['m']; tot[q][1] += w['m'] * w['coverage'][q]
    if len(d['windows']) >= 3:
        x = np.log([w['N0'] for w in d['windows']]); y = np.log([w['median_Y'] for w in d['windows']]) - x / 3 * 0 
        # median nu_min = Y * N0^(-1/3)  ->  log median nu_min = log Y - (1/3) log N0
        b = np.polyfit(x, y - x / 3, 1)[0]; slopes.append((d['name'], float(b)))
cov = {q: (v[1] / v[0], v[0]) for q, v in tot.items()}
print("pooled coverage:", cov, " expected 0.95 / 0.99")
print("slopes d log(median nu_min)/d log N0 (theory -1/3):")
for s in slopes: print("  ", s)
# extrapolated window times for all zeros in [x,2x]
ext = []
for X in (1e9, 1e11, 1e13, 1e15, 1e18, 1e20, 1e22):
    L = np.log(X / (2 * np.pi)); N = X / (2 * np.pi) * (L + 2 * np.log(2) - 1)
    d = dict(x=X, N=N, L=L)
    for q in (0.05, 0.5):
        nq = brentq(lambda y: N * F(y) + np.log(1 - q), 1e-14, 0.6); d[f"tau_q{q}"] = float(2 * np.pi**2 / L**2 * nq**2)
    ext.append(d); print(f"x={X:.0e} N={N:.2e} tau_5%={d['tau_q0.05']:.2e} tau_50%={d['tau_q0.5']:.2e}")
json.dump(dict(rows=rows, coverage={k: list(v) for k, v in cov.items()}, slopes=slopes, extrapolation=ext), open('final_stats.json', 'w'), indent=1)
