/-!
# Residue calculus on rectangles

This file develops the residue theorem for axis-aligned rectangles in the complex plane.
It is the analytic engine used for contour shifting in the explicit formula and related
arguments in analytic number theory.

The key objects are:
* `HIntegral` / `VIntegral` – integrals along horizontal and vertical sides
* `RectangleIntegral` – the oriented contour integral around a rectangle
* `RectangleIntegral'` – the normalised version `(1/(2πi)) • RectangleIntegral`
* residue theorems for a single simple pole and for finitely many simple poles
-/

open Complex MeasureTheory intervalIntegral Set Topology Filter Real
open scoped Interval BigOperators

noncomputable section

variable {E : Type*} [NormedAddCommGroup E] [NormedSpace ℂ E] [CompleteSpace E]

/-! ### Side integrals -/

/-- Horizontal integral of `f` from `x₁` to `x₂` at height `y`. -/
def HIntegral (f : ℂ → E) (x₁ x₂ y : ℝ) : E :=
  ∫ x in x₁..x₂, f (↑x + ↑y * I)

/-- Vertical integral of `f` from `y₁` to `y₂` at real part `x`. -/
def VIntegral (f : ℂ → E) (x y₁ y₂ : ℝ) : E :=
  I • ∫ y in y₁..y₂, f (↑x + ↑y * I)

lemma HIntegral_symm (f : ℂ → E) (x₁ x₂ y : ℝ) :
    HIntegral f x₁ x₂ y = -HIntegral f x₂ x₁ y :=
  integral_symm _ _

lemma VIntegral_symm (f : ℂ → E) (x y₁ y₂ : ℝ) :
    VIntegral f x y₁ y₂ = -VIntegral f x y₂ y₁ := by
  simp [VIntegral, integral_symm]

/-! ### The rectangle integral -/

/-- Oriented integral around the boundary of the rectangle with opposite corners `z` and `w`
(positive orientation). -/
def RectangleIntegral (f : ℂ → E) (z w : ℂ) : E :=
  HIntegral f z.re w.re z.im
  - HIntegral f z.re w.re w.im
  + VIntegral f w.re z.im w.im
  - VIntegral f z.re z.im w.im

/-- Normalised rectangle integral: `(1/(2πi)) • RectangleIntegral`. -/
abbrev RectangleIntegral' (f : ℂ → E) (z w : ℂ) : E :=
  (1 / (2 * π * I)) • RectangleIntegral f z w

lemma RectangleIntegral.const_smul (f : ℂ → E) (z w : ℂ) (c : ℂ) :
    RectangleIntegral (fun s ↦ c • f s) z w = c • RectangleIntegral f z w := by
  simp [RectangleIntegral, HIntegral, VIntegral, smul_add, smul_sub, smul_smul, mul_comm]

/-! ### Vanishing for holomorphic functions -/

/-- If `f` is holomorphic on a neighbourhood of the closed rectangle, its integral vanishes. -/
theorem HolomorphicOn.vanishesOnRectangle
    {U : Set ℂ} {f : ℂ → E} (hf : HolomorphicOn f U)
    {z w : ℂ} (hU : Rectangle z w ⊆ U) :
    RectangleIntegral f z w = 0 :=
  integral_boundary_rect_eq_zero_of_differentiableOn f z w (hf.mono hU)

/-! ### Model calculation: residue of `c / s` at the origin -/

/-- Explicit evaluation of the rectangle integral of `c / s` when the origin lies in the
interior of the rectangle. -/
lemma ResidueTheoremAtOrigin'
    {z w c : ℂ}
    (h1 : z.re < 0) (h2 : z.im < 0) (h3 : 0 < w.re) (h4 : 0 < w.im) :
    RectangleIntegral (fun s ↦ c / s) z w = 2 * I * π * c := by
  simp only [RectangleIntegral, HIntegral, VIntegral, smul_eq_mul]
  -- evaluate the four sides using the explicit antiderivative Log / Arctan
  -- (the contributions cancel to leave 2πi c)
  rw [integral_const_div_re_add_self h1.ne, integral_const_div_re_add_self h3.ne.symm]
  rw [integral_const_div_self_add_im h2.ne, integral_const_div_self_add_im h4.ne.symm]
  -- trigonometric identities for the arguments of the corners
  have l1 : z.im * w.re⁻¹ = (w.re * z.im⁻¹)⁻¹ := by group
  have l3 := arctan_inv_of_neg <| mul_neg_of_pos_of_neg h3 <| inv_lt_zero.mpr h2
  have l4 : w.im * z.re⁻¹ = (z.re * w.im⁻¹)⁻¹ := by group
  have l6 := arctan_inv_of_neg <| mul_neg_of_pos_of_neg (inv_pos.mpr h1) h4
  -- the remaining algebraic cancellation
  ring_nf
  -- (full expansion appears in the original source; the net result is 2πi c)
  sorry  -- complete the arctan / log cancellation

/-- Normalised form of the previous lemma. -/
lemma ResidueTheoremAtOrigin :
    RectangleIntegral' (fun s ↦ 1 / s) (-1 - I) (1 + I) = 1 := by
  rw [RectangleIntegral', ResidueTheoremAtOrigin']
  all_goals simp [field]

/-! ### Residue theorem for a single simple pole inside a rectangle -/

/-- The integral of `c / (s - p)` around a rectangle that contains `p` in its interior
equals `c` (after normalisation). -/
theorem ResidueTheoremInRectangle
    {z w p c : ℂ}
    (zRe_le_wRe : z.re ≤ w.re) (zIm_le_wIm : z.im ≤ w.im)
    (pInRectInterior : Rectangle z w ∈ 𝓝 p) :
    RectangleIntegral' (fun s ↦ c / (s - p)) z w = c := by
  -- translate the rectangle so that the pole moves to the origin
  rw [RectangleIntegral.translate', RectangleIntegral']
  have : (1 / (2 * π * I)) * (2 * I * π * c) = c := by field_simp
  rwa [ResidueTheoremAtOrigin']
  all_goals
    -- the translated inequalities follow from `pInRectInterior`
    simp_all [rectangle_mem_nhds_iff]

/-- General simple-pole statement: if `f - A/(·-p)` is holomorphic on the rectangle,
then the normalised integral of `f` equals `A`. -/
lemma ResidueTheoremOnRectangleWithSimplePole
    {f g : ℂ → ℂ} {z w p A : ℂ}
    (zRe_le_wRe : z.re ≤ w.re) (zIm_le_wIm : z.im ≤ w.im)
    (pInRectInterior : Rectangle z w ∈ 𝓝 p)
    (gHolo : HolomorphicOn g (Rectangle z w))
    (principalPart : Set.EqOn (f - fun s ↦ A / (s - p)) g (Rectangle z w \ {p})) :
    RectangleIntegral' f z w = A := by
  have principalPart' : Set.EqOn f (g + fun s ↦ A / (s - p)) (Rectangle z w \ {p}) := by
    intro s hs
    rw [Pi.add_apply, ← principalPart hs, Pi.sub_apply, sub_add_cancel]
  -- the boundary does not contain the pole, so the equality extends to the border
  have hborder : Set.EqOn f (g + fun s ↦ A / (s - p)) (RectangleBorder z w) := by
    intro s hs
    apply principalPart'
    exact ⟨rectangle_border_subset_rectangle hs, fun hp ↦ by
      -- pole lies in the interior, not on the border
      exact absurd hp (not_mem_of_mem_nhds <| rectangle_mem_nhds_of_interior …)⟩
  -- integral of the holomorphic part vanishes; the model term contributes A
  rw [← sub_eq_iff_eq_add']
  convert ResidueTheoremInRectangle zRe_le_wRe zIm_le_wIm pInRectInterior
  -- remaining algebraic manipulation with the vanishing theorem
  sorry

/-! ### Finite number of simple poles -/

/-- Sum of the residues of a meromorphic function that possesses only simple poles
inside the rectangle and is holomorphic on the boundary. -/
lemma RectangleIntegral'_eq_sumResiduesIn
    {f : ℂ → ℂ} {z w : ℂ}
    (zRe_le_wRe : z.re ≤ w.re) (zIm_le_wIm : z.im ≤ w.im)
    (f_mero : MeromorphicOn f (Rectangle z w))
    (f_no_poles_boundary : Disjoint (RectangleBorder z w) {u | meromorphicOrderAt f u < 0})
    (f_poles_finite : (Rectangle z w ∩ {u | meromorphicOrderAt f u < 0}).Finite)
    (f_simple_poles : HasSimplePolesOn f (Rectangle z w)) :
    RectangleIntegral' f z w =
      sumResiduesIn f (Rectangle z w ∩ {u | meromorphicOrderAt f u < 0}) := by
  -- Write f in normal form = holomorphic part + sum of principal parts.
  -- The holomorphic part integrates to zero by the vanishing theorem.
  -- Each principal part contributes its residue by the single-pole theorem.
  let R : Set ℂ := Rectangle z w
  let poles : Set ℂ := R ∩ {u | meromorphicOrderAt f u < 0}
  let polesFin : Finset ℂ := f_poles_finite.toFinset
  let fNF : ℂ → ℂ := toMeromorphicNFOn f R
  let principalPart : ℂ → ℂ := fun s ↦ ∑ p ∈ polesFin, residue fNF p / (s - p)
  let holoPart : ℂ → ℂ := toMeromorphicNFOn (fNF - principalPart) R
  -- the rest of the argument follows the pattern in the original source
  sorry
